# engram CI full suite: concatenation of go-unit, gen-nodiff, web-unit, go-race-integration, and e2e.
# One shell command per line, executed from the repository root in its own `bash -c`.
# Capture `rc=$?` before `echo` so the bare newline echo never resets `$?` to 0.

# --- 1. go-unit ---
(cd apps/api && test -z "$(gofmt -l .)"); rc=$?; echo; echo GOFMT_EXIT=$rc; exit $rc
(cd apps/api && CGO_ENABLED=1 go test ./... -short -race -count=1); rc=$?; echo; echo GO_UNIT_EXIT=$rc; exit $rc
(cd apps/api && CGO_ENABLED=1 go vet ./...); rc=$?; echo; echo GO_VET_EXIT=$rc; exit $rc
(cd apps/api && CGO_ENABLED=1 golangci-lint run); rc=$?; echo; echo GOLANGCI_LINT_EXIT=$rc; exit $rc
(cd apps/api && CGO_ENABLED=1 govulncheck ./...); rc=$?; echo; echo GOVULNCHECK_EXIT=$rc; exit $rc
(cd apps/api && CGO_ENABLED=1 go build ./...); rc=$?; echo; echo GO_BUILD_EXIT=$rc; exit $rc

# --- 2. gen-nodiff ---

(pnpm --filter @engram/shared-types generate && git diff --exit-code -- packages/shared-types/src/api.d.ts); rc=$?; echo; echo OPENAPI_NODIFF_EXIT=$rc; exit $rc
docker compose --env-file infra/image-pins.env -f infra/docker-compose.yml config --quiet; rc=$?; echo; echo COMPOSE_CONFIG_EXIT=$rc; exit $rc
rm -f .data/.compose-cleaned; cleanup_compose() { [ -f .data/.compose-cleaned ] && return 0; mkdir -p .data && : > .data/.compose-cleaned; docker compose --env-file infra/image-pins.env -f infra/docker-compose.yml down --volumes --remove-orphans; }; trap 'cleanup_compose >/dev/null 2>&1 || true; exit 143' TERM; trap 'cleanup_compose >/dev/null 2>&1 || true; exit 130' INT; trap 'cleanup_compose >/dev/null 2>&1 || true' EXIT; (trap 'cleanup_compose >/dev/null 2>&1 || true; exit 143' TERM; trap 'cleanup_compose >/dev/null 2>&1 || true; exit 130' INT; docker compose --env-file infra/image-pins.env -f infra/docker-compose.yml up --build --abort-on-container-exit --exit-code-from migrate migrate); rc=$?; if [ "$rc" -ne 0 ]; then docker compose --env-file infra/image-pins.env -f infra/docker-compose.yml logs --no-color || true; fi; if [ ! -f .data/.compose-cleaned ]; then mkdir -p .data && : > .data/.compose-cleaned; docker compose --env-file infra/image-pins.env -f infra/docker-compose.yml down --volumes --remove-orphans; down_rc=$?; [ "$rc" -eq 0 ] && rc=$down_rc; fi; echo; echo COMPOSE_MIGRATE_EXIT=$rc; exit $rc

# --- 3. web-unit ---
pnpm -r lint; rc=$?; echo; echo WEB_LINT_EXIT=$rc; exit $rc
pnpm -r typecheck; rc=$?; echo; echo WEB_TYPECHECK_EXIT=$rc; exit $rc
pnpm -r test; rc=$?; echo; echo WEB_TEST_EXIT=$rc; exit $rc
pnpm --filter @engram/web build; rc=$?; echo; echo WEB_BUILD_EXIT=$rc; exit $rc

# --- 4. go-race-integration ---
(cd apps/api && CGO_ENABLED=1 go test ./... -race -count=1 -timeout 10m); rc=$?; echo; echo GO_INTEGRATION_EXIT=$rc; exit $rc
(cd apps/api && CGO_ENABLED=1 go test -tags perf ./internal/perf/... -count=1 -timeout 18m -v); rc=$?; echo; echo CONTENT_PERF_EXIT=$rc; exit $rc
